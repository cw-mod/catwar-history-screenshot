   117→   117→    function drawClockStampOnCanvas(ctx, scale, targetRect, h2Rect) {
   118→   118→        const msk = getMoscowDate();
   119→   119→        const timeStr = `${pad(msk.getHours())}:${pad(msk.getMinutes())}:${pad(msk.getSeconds())}`;
   120→   120→        const dateStr = `${pad(msk.getDate())}.${pad(msk.getMonth() + 1)}.${String(msk.getFullYear()).slice(-2)}`;
   121→   121→
   122→   122→        // Размеры шрифтов (в canvas пикселях)
   123→   123→        const mainFont = Math.max(20, 28 * scale);
   124→   124→        const iconFont = Math.max(14, 18 * scale);
   125→   125→        const mskFont = Math.max(11, 14 * scale);   // MSK меньше основного шрифта
   126→   126→
   127→   127→        const mainText = `${timeStr} ${dateStr}`;
   128→   128→        const iconChar = '\u{1F30D}\u{FE0E}';   // 🌍 + U+FE0E (монохромная)
   129→   129→
   130→   130→        // Замеры
   131→   131→        ctx.font = `${mainFont}px Arial`;
   132→   132→        const mainW = ctx.measureText(mainText).width;
   133→   133→
   134→   134→        ctx.font = `${iconFont}px Arial`;
   135→   135→        const iconW = ctx.measureText(iconChar).width;
   136→   136→
   137→   137→        ctx.font = `${mskFont}px Arial`;
   138→   138→        const mskW = ctx.measureText('MSK').width;
   139→   139→
   140→   140→        // Layout бокса
   141→   141→        const boxPad = 5 * scale;
   142→   142→        const gap = 6 * scale;          // между mainText и правой колонкой
   143→   143→        const smallGap = 1 * scale;     // между 🌍 и MSK
   144→   144→        const rightColW = Math.max(iconW, mskW);
   145→   145→        const rightColH = iconFont + smallGap + mskFont;
   146→   146→        const contentH = Math.max(mainFont, rightColH);
   147→   147→
   148→   148→        const boxW = boxPad + mainW + gap + rightColW + boxPad;
   149→   149→        const boxH = boxPad + contentH + boxPad;
   150→   150→
   151→   151→        // Бейдж рядом с текстом «История» — справа от <a> внутри h2.
   152→   152→        // Вертикально — по центру h2.
   153→   153→        let boxX, boxY;
   154→   154→        if (h2Rect) {
   155→   155→            const h2RightOnCanvas = (h2Rect.right - targetRect.left) * scale;
   156→   156→            const h2CenterYOnCanvas = ((h2Rect.top + h2Rect.bottom) / 2 - targetRect.top) * scale;
   157→   157→            boxX = h2RightOnCanvas + 8 * scale;
   158→   158→            boxY = h2CenterYOnCanvas - boxH / 2;
   159→   159→        } else {
   160→   160→            boxX = ctx.canvas.width - boxW - 10 * scale;
   161→   161→            boxY = 10 * scale;
   162→   162→        }
   163→   163→
   164→   164→        console.info('[CatWar Screenshot] drawClockStampOnCanvas:', {
   165→   165→            canvasW: ctx.canvas.width,
   166→   166→            canvasH: ctx.canvas.height,
   167→   167→            boxX: boxX,
   168→   168→            boxY: boxY,
   169→   169→            boxW: boxW,
   170→   170→            boxH: boxH,
   171→   171→            mainW: mainW,
   172→   172→            iconW: iconW,
   173→   173→            mskW: mskW,
   174→   174→            scale: scale,
   175→   175→            h2Rect: h2Rect ? 'yes' : 'null'
   176→   176→        });
   177→   177→
   178→   178→        // Заливка фона со скруглёнными углами
   179→   179→        const r = Math.min(10 * scale, boxW / 2, boxH / 2);
   180→   180→        ctx.fillStyle = CLOCK_BG;
   181→   181→        ctx.beginPath();
   182→   182→        ctx.moveTo(boxX + r, boxY);
   183→   183→        ctx.lineTo(boxX + boxW - r, boxY);
   184→   184→        ctx.quadraticCurveTo(boxX + boxW, boxY, boxX + boxW, boxY + r);
   185→   185→        ctx.lineTo(boxX + boxW, boxY + boxH - r);
   186→   186→        ctx.quadraticCurveTo(boxX + boxW, boxY + boxH, boxX + boxW - r, boxY + boxH);
   187→   187→        ctx.lineTo(boxX + r, boxY + boxH);
   188→   188→        ctx.quadraticCurveTo(boxX, boxY + boxH, boxX, boxY + boxH - r);
   189→   189→        ctx.lineTo(boxX, boxY + r);
   190→   190→        ctx.quadraticCurveTo(boxX, boxY, boxX + r, boxY);
   191→   191→        ctx.closePath();
   192→   192→        ctx.fill();
   193→   193→
   194→   194→        // Текст: чёрный
   195→   195→        ctx.fillStyle = CLOCK_TEXT;
   196→   196→
   197→   197→        const contentTop = boxY + boxPad;
   198→   198→        const contentCenterY = contentTop + contentH / 2;

   215→   215→
   216→   216→        // MSK (знаменатель, меньше)
   217→   217→        ctx.font = `${mskFont}px Arial`;

   219→   219→
   220→   220→        // Восстанавливаем ctx

   222→   222→        ctx.textBaseline = 'alphabetic';

   253→   253→    // =========================================================================
   254→   254→
   255→   255→    // В клонированном документе Multi-Find не активен (его JS не работает
   256→   256→    // в клоне), поэтому безопасно можно:
   257→   257→    //   1. Заменить <multi-find-1-extension> на набор <span> по одному на
   258→   258→    //      каждое слово (display: inline-block) + <span> с пробелами между
   259→   259→    //      ними (display: inline-block, transparent). Иначе html2canvas
   260→   260→    //      рендерит подряд идущие inline-span с одинаковым фоном как один
   261→   261→    //      сплошной блок — подсветка «слипается».
   262→   262→    //   2. Санировать color()/oklch()/oklab()/lab()/lch()/hwb() в computed
   263→   263→    //      стилях — html2canvas их не умеет парсить и падает.
   264→   264→    function sanitizeClone(clonedDoc, rootEl) {
   265→   265→        const win = clonedDoc.defaultView;
   266→   266→        let replacedCount = 0;
   267→   267→
   268→   268→        // --- 1. Замена multi-find-extension на набор слов в inline-block ---
   269→   269→        const mfeTargets = Array.from(rootEl.querySelectorAll('multi-find-1-extension'));
   270→   270→        mfeTargets.forEach(original => {
   271→   271→            let inlineBg = original.style.backgroundColor;
   272→   272→            let inlineColor = original.style.color;
   273→   273→
   274→   274→            // Если инлайн-стиля нет — берём из computed
   275→   275→            if ((!inlineBg || !inlineColor) && win) {
   276→   276→                try {
   277→   277→                    const cs = win.getComputedStyle(original);
   278→   278→                    if (!inlineBg) inlineBg = cs.backgroundColor;
   279→   279→                    if (!inlineColor) inlineColor = cs.color;
   280→   280→                } catch (e) {}
   281→   281→            }
   282→   282→
   283→   283→            // inline-block: имеет явную ширину по содержимому, не растягивается
   284→   284→            // на всю строку и не сливается с соседями.
   285→   285→            const wordCss =
   286→   286→                `background-color: ${inlineBg || 'transparent'}; ` +
   287→   287→                `color: ${inlineColor || 'inherit'}; ` +
   288→   288→                `display: inline-block; ` +
   289→   289→                `padding: 0; ` +
   290→   290→                `margin: 0; ` +
   291→   291→                `line-height: inherit;`;
   292→   292→
   293→   293→            // Пробелы — отдельный span с transparent фоном и inline-block,
   294→   294→            // чтобы html2canvas точно видел разделитель между подсвеченными
   295→   295→            // словами (text node может рендериться с нулевой шириной).
   296→   296→            const spaceCss =
   297→   297→                `display: inline-block; ` +
   298→   298→                `background: transparent;`;
   299→   299→
   300→   300→            const fragment = clonedDoc.createDocumentFragment();
   301→   301→            const text = original.textContent || '';
   302→   302→            const parts = text.split(/(\s+)/);
   303→   303→            parts.forEach(part => {
   304→   304→                if (!part) return;
   305→   305→                if (/^\s+$/.test(part)) {
   306→   306→                    const spaceSpan = clonedDoc.createElement('span');
   307→   307→                    spaceSpan.style.cssText = spaceCss;
   308→   308→                    spaceSpan.textContent = part;
   309→   309→                    fragment.appendChild(spaceSpan);
   310→   310→                } else {
   311→   311→                    const span = clonedDoc.createElement('span');
   312→   312→                    span.style.cssText = wordCss;
   313→   313→                    span.textContent = part;
   314→   314→                    fragment.appendChild(span);
   315→   315→                }
   316→   316→            });
   317→   317→
   318→   318→            if (original.parentNode) {
   319→   319→                original.parentNode.replaceChild(fragment, original);
   320→   320→                replacedCount++;
   321→   321→            }
   322→   322→        });
   323→   323→
   324→   324→        console.info('[CatWar Screenshot] sanitizeClone: заменено ' + replacedCount + ' multi-find элементов');
   325→   325→
   326→   326→        // --- 1b. Подменяем фон td#history на непрозрачный rgb(107,86,88) ---
   327→   327→        // В CSS игры фон задан как rgba(...,0.7), html2canvas рендерит поверх
   328→   328→        // белого и получает светлый rgb(172,158,162). Подменяем на непрозрачный
   329→   329→        // тёмный, как в игре.
   330→   330→        rootEl.style.setProperty('background-color', TARGET_BG_COLOR, 'important');
   331→   331→
   332→   332→        // --- 2. Санирование color-функций ---
   333→   333→        if (!win) return;
   334→   334→        const elements = [rootEl, ...rootEl.querySelectorAll('*')];
   335→   335→        elements.forEach(el => {
   336→   336→            let cs;
   337→   337→            try { cs = win.getComputedStyle(el); } catch (e) { return; }
   338→   338→
   339→   339→            COLOR_PROPS_KEBAB.forEach(prop => {
   340→   340→                let val;
   341→   341→                try { val = cs.getPropertyValue(prop); } catch (e) { return; }
   342→   342→                if (!val) return;
   343→   343→                const lower = val.toLowerCase();
   344→   344→                if (!UNSUPPORTED_COLOR_FUNCS.some(p => lower.includes(p))) return;
   345→   345→
   346→   346→                let safe;
   347→   347→                if (prop === 'background-color') {
   348→   348→                    safe = 'rgba(184, 170, 170, 0.55)';
   349→   349→                } else if (prop === 'box-shadow') {
   350→   350→                    safe = 'none';
   351→   351→                } else {
   352→   352→                    safe = CLOCK_BG;
   353→   353→                }
   354→   354→                el.style.setProperty(prop, safe, 'important');
   355→   355→            });
   356→   356→        });
   357→   357→    }

   363→   363→    function downloadCanvas(canvas, modeSlug) {
   364→   364→        const msk = getMoscowDate();
   365→   365→        const stamp =
   366→   366→            `${pad(msk.getDate())}.${pad(msk.getMonth() + 1)}.${msk.getFullYear()}_` +
   367→   367→            `${pad(msk.getHours())}-${pad(msk.getMinutes())}-${pad(msk.getSeconds())}`;
   368→   368→        const filename = `история_${modeSlug}_${stamp}_МСК.png`;
   369→   369→
   370→   370→        canvas.toBlob(function (blob) {
   371→   371→            if (!blob) {
   372→   372→                alert('Не удалось собрать PNG (canvas пустой).');
   373→   373→                return;
   374→   374→            }
   375→   375→            const url = URL.createObjectURL(blob);
   376→   376→            const a = document.createElement('a');
   377→   377→            a.href = url;
   378→   378→            a.download = filename;
   379→   379→            document.body.appendChild(a);
   380→   380→            a.click();
   381→   381→            document.body.removeChild(a);
   382→   382→            setTimeout(() => URL.revokeObjectURL(url), 2000);
   383→   383→        }, 'image/png');
   384→   384→    }

   503→   503→        function finalize(rawCanvas) {
   504→   504→            cleanup();
   505→   505→
   506→   506→            // Для visible режима — обрезаем canvas до видимой области.
   507→   507→            // После раскрытия canvas содержит полный td#history: padding.top +
   508→   508→            // h2 + полная #history_block (без скролла).
   509→   509→            // Хотим показать: padding.top + h2 + видимую часть #history_block
   510→   510→            // (с учётом scrollTop).
   511→   511→            let finalCanvas = rawCanvas;
   512→   512→            if (mode === 'visible') {
   513→   513→                try {
   514→   514→                    const s = scale;
   515→   515→                    const upperEnd = Math.round((h2OffsetInTarget + h2Height) * s);
   516→   516→                    const histSrcY = Math.round((h2OffsetInTarget + h2Height + scrollTop) * s);
   517→   517→                    const histSrcH = Math.max(0, Math.round((visibleClientHeight - h2OffsetInTarget - h2Height) * s));
   518→   518→
   519→   519→                    const newCanvas = document.createElement('canvas');
   520→   520→                    newCanvas.width = rawCanvas.width;
   521→   521→                    newCanvas.height = Math.round(visibleClientHeight * s);
   522→   522→                    const ctx = newCanvas.getContext('2d');
   523→   523→
   524→   524→                    ctx.fillStyle = '#ffffff';
   525→   525→                    ctx.fillRect(0, 0, newCanvas.width, newCanvas.height);
   526→   526→
   527→   527→                    if (upperEnd > 0) {
   528→   528→                        ctx.drawImage(rawCanvas,
   529→   529→                            0, 0, rawCanvas.width, upperEnd,
   530→   530→                            0, 0, rawCanvas.width, upperEnd);
   531→   531→                    }
   532→   532→
   533→   533→                    if (histSrcH > 0) {
   534→   534→                        ctx.drawImage(rawCanvas,
   535→   535→                            0, histSrcY, rawCanvas.width, histSrcH,
   536→   536→                            0, upperEnd, rawCanvas.width, histSrcH);
   537→   537→                    }
   538→   538→
   539→   539→                    finalCanvas = newCanvas;
   540→   540→                } catch (e) {
   541→   541→                    console.warn('[CatWar Screenshot] Обрезка visible упала:', e);
   542→   542→                }
   543→   543→            } else {
   544→   544→                // Для full режима — копируем rawCanvas на новый canvas,
   545→   545→                // потому что fillText может не работать на canvas от html2canvas.
   546→   546→                try {
   547→   547→                    const copyCanvas = document.createElement('canvas');
   548→   548→                    copyCanvas.width = rawCanvas.width;
   549→   549→                    copyCanvas.height = rawCanvas.height;
   550→   550→                    const copyCtx = copyCanvas.getContext('2d');
   551→   551→                    copyCtx.drawImage(rawCanvas, 0, 0);
   552→   552→                    finalCanvas = copyCanvas;
   553→   553→                } catch (e) {
   554→   554→                    console.warn('[CatWar Screenshot] Копирование full canvas упало:', e);
   555→   555→                }

   559→   559→            // или на rawCanvas для full). Html2canvas не умеет двухуровневый
   560→   560→            // layout — поэтому рисуем сами через ctx.fillText.
   561→   561→            // Позиция — справа от текста «История» (<a> внутри h2).
   562→   562→            try {

   575→   575→            }, 1500);
   576→   576→        }
